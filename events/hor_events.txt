add_namespace = hor_events

# Yugoslavia asks to join Axis
country_event = {
	id = hor_events.1
	title = hor_events.1.t
	desc = hor_events.1.d

	picture = GFX_report_event_generic_read_write
	
	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			GER = {
				save_event_target_as = alliance_inviter
			}
			YUG = {
				save_event_target_as = alliance_applicant
			}
		}
	}

	# Accept them
	option = {
		name = hor_events.1.a
		trigger = {
			YUG = { has_rule = can_join_factions }
		}
		ai_chance = {
			base = 100
		}
		add_to_faction = YUG
		YUG = { country_event = { id = generic.3 hours = 6 } }
		hidden_effect = {
			YUG = { army_experience = 35 }
			YUG = { add_war_support = 0.1 }
			YUG = { set_politics = {
    					ruling_party = fascism
    					elections_allowed = no
					}
			}
			YUG = { set_political_party = {
    					ideology = fascism
    					popularity = 100
					}
			}
			set_country_flag = alliance_inviter@ROOT
			YUG = { set_country_flag = alliance_applicant@ROOT }
			news_event = { id = news.289 hours = 12 }
		}

		YUG = {
			add_ai_strategy = {
				type = alliance
				id = GER
				value = 200
			}
		}
	}
	# Decline them
	option = { 
		name = hor_events.1.b
		ai_chance = {
			base = 0
		}
		YUG = { 
			country_event = { id = generic.4 hours = 6 }
		}
	}		
}

# Japan requests to puppet us (Siam)
country_event = {
	id = hor_events.2
	title = hor_events.2.t
	desc = hor_events.2.d

	picture = GFX_report_event_japanese_siam_politicians
	
	is_triggered_only = yes

	# Accept their offer (Siam gets puppeted)
	option = {
		name = hor_events.2.a
		SIA = { set_politics = { ruling_party = JAP } }
		#puppet = SIA
		
		SIA_phraya_phahon = {
			add_country_leader_role = {
				country_leader = {
					expire = "1965.1.1"
					ideology = fascism_ideology
					traits = { collaborator_king }
				}
				promote_leader = yes
			}
		}

		JAP = {
			country_event = { id = hor_events.3 }

			add_to_faction = SIA

			set_autonomy = {
    				target = SIA
				freedom_level = 0.1
    				autonomous_state = hor_siam_autonomy
    				end_wars = no
    				end_civil_wars = no
			}
		}
	}

	# Decline their offer (Siam doesn't get puppeted)
	option = { 
		name = hor_events.2.b
		ai_chance = {
			base = 0
		}
		JAP = { 
			country_event = { id = hor_events.4 }
		}
	}
}
